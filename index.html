async function connect() { try { if (!navigator.bluetooth) { log("Web Bluetooth not supported in this browser."); return; } if (!window.isSecureContext) { log("ERROR: page is not a secure context. Use https:// or http://localhost"); return; } log("Secure context OK. Opening chooser..."); // Debug mode: show ALL Bluetooth LE devices device = await navigator.bluetooth.requestDevice({ acceptAllDevices: true, optionalServices: [SERVICE_UUID] }); log("Device selected: " + device.name); device.addEventListener("gattserverdisconnected", onDisconnect); log("Connecting GATT..."); server = await device.gatt.connect(); log("GATT connected"); service = await server.getPrimaryService(SERVICE_UUID); log("Primary service found"); timeChar = await service.getCharacteristic(TIME_UUID); ctrlChar = await service.getCharacteristic(CTRL_UUID); imuChar = await service.getCharacteristic(IMU_UUID); micChar = await service.getCharacteristic(MIC_UUID); log("Characteristics found"); // Sync Unix epoch ms const now = BigInt(Date.now()); const buf = new ArrayBuffer(8); const dv = new DataView(buf); dv.setBigUint64(0, now, true); await timeChar.writeValue(buf); log("Time synced: " + now.toString()); await imuChar.startNotifications(); await micChar.startNotifications(); imuChar.addEventListener("characteristicvaluechanged", onImu); micChar.addEventListener("characteristicvaluechanged", onMic); statusEl.textContent = "connected"; statusEl.classList.add("on"); $("btnStart").disabled = false; $("btnDownloadImu").disabled = false; $("btnDownloadMic").disabled = false; log("Ready."); } catch (error) { log(error.name + ": " + error.message); if (error.name === "NotFoundError") { log("Chooser was cancelled or no device selected."); } if (error.name === "SecurityError") { log("Web Bluetooth blocked. Use HTTPS or localhost."); } if (error.name === "NetworkError") { log("Bluetooth radio unavailable or permission denied."); } } }
